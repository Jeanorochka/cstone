cmake_minimum_required(VERSION 3.20)
project(CstoneConsumer LANGUAGES C CXX)
find_package(Cstone 0.2 REQUIRED CONFIG)
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE Cstone::cstone)
add_executable(consumer_cpp main.cpp)
target_compile_features(consumer_cpp PRIVATE cxx_std_11)
target_link_libraries(consumer_cpp PRIVATE Cstone::cstone)
